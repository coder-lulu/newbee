// 1. JWT缓存分片优化 - 预期性能提升40%
func EnterpriseGrade(jwtSecret string) *OptimalAuth {
    return &OptimalConfig{
        Performance: PerformanceOpts{
            CacheSize:   100000,    // 从50000增加到100000

            CacheTTL:    15 * time.Minute, // 从10分钟增加到15分钟
        },
    }
}

// 2. 数据权限L1缓存扩容 - 预期命中率从80%提升到90%
l1Config := &L1CacheConfig{
    MaxSize:     5000,                // 从2000增加到5000
    DefaultTTL:  5 * time.Minute,     // 从3分钟增加到5分钟
    MaxMemoryMB: 128,                 // 从64MB增加到128MB
}

// 3. Redis连接池优化 - 预期吞吐量提升60%
redisConfig := &OptimizedRedisConfig{
    PoolSize:     200,  // 从50增加到200

}
